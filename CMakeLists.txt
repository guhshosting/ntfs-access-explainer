cmake_minimum_required(VERSION 3.24)
project(ntfs-access-explainer VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows and MSVC are required")
endif()
option(NAE_ANALYZE "Enable MSVC static analysis" OFF)
option(NAE_ASAN "Enable AddressSanitizer" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(nae_core src/core.cpp)
target_include_directories(nae_core PUBLIC src)
target_compile_definitions(nae_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(nae_core PUBLIC advapi32 authz)
add_executable(ntfs-access-explainer src/main.cpp)
target_link_libraries(ntfs-access-explainer PRIVATE nae_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(nae_tests tests/tests.cpp)
  target_link_libraries(nae_tests PRIVATE nae_core)
  add_test(NAME security_and_fixtures COMMAND nae_tests)
  add_test(NAME help COMMAND ntfs-access-explainer --help)
  add_test(NAME version COMMAND ntfs-access-explainer --version)
  find_program(NAE_PWSH pwsh)
  if(NAE_PWSH)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/json-artifacts")
    set_tests_properties(security_and_fixtures PROPERTIES ENVIRONMENT "NAE_TEST_ARTIFACTS=${CMAKE_CURRENT_BINARY_DIR}/json-artifacts" FIXTURES_SETUP json_reports)
    add_test(NAME independent_json_schema COMMAND "${NAE_PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tools/validate-json.ps1" -Artifacts "${CMAKE_CURRENT_BINARY_DIR}/json-artifacts")
    set_tests_properties(independent_json_schema PROPERTIES FIXTURES_REQUIRED json_reports)
  endif()
endif()
foreach(t nae_core ntfs-access-explainer nae_tests)
  if(TARGET ${t})
    target_compile_options(${t} PRIVATE /W4 /WX /utf-8 /permissive- /EHsc)
    if(NAE_ANALYZE)
      target_compile_options(${t} PRIVATE /analyze)
    endif()
    if(NAE_ASAN)
      target_compile_options(${t} PRIVATE /fsanitize=address /Zi)
      target_link_options(${t} PRIVATE /INCREMENTAL:NO)
    endif()
  endif()
endforeach()
install(TARGETS ntfs-access-explainer RUNTIME DESTINATION bin)
install(FILES README.md LICENSE VALIDATION.md DESTINATION share/ntfs-access-explainer)
install(FILES docs/schema-v1.json DESTINATION share/ntfs-access-explainer/docs)
